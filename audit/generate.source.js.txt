const fs = require("fs");
const path = require("path");

const ROOT = path.resolve(__dirname, "..");
const DATA_PATH = path.join(ROOT, "data", "planetary-boundaries.json");
const SVG_DIR = path.join(ROOT, "svg");
const INDEX_PATH = path.join(ROOT, "index.html");

const PT = 0.352777778;
const STROKE_MAIN = 0.5 * PT;
const STROKE_AUX = 0.25 * PT;
const FONT = {
  title: 3.0,
  zone: 2.45,
  marker: 2.45,
  note: 2.55,
  unresolved: 2.55,
  footer: 2.05
};

const STYLE = {
  safeLine: "#DDD8D4",
  exceedLine: "#DE6C58",
  text: "#000000",
  background: "#FFFFFF"
};

const PAGE = { width: 210, height: 280 };
const LAYOUT = {
  lineX1: 38,
  lineX2: 176,
  boundaryX: 96,
  highRiskX: 132,
  titleY: 176,
  zoneLabelY: 193.5,
  lineY: 202,
  tickHalf: 2.8,
  markerLabelY: 211.8,
  markerLabelGap: 4.7,
  noteX: 82,
  noteY: 231.5,
  noteLineGap: 4.9,
  footerX: 38,
  footerY: 260.8,
  footerLineGap: 4.1
};

function readData() {
  return JSON.parse(fs.readFileSync(DATA_PATH, "utf8"));
}

function ensureDir(dir) {
  fs.mkdirSync(dir, { recursive: true });
}

function esc(value) {
  return String(value)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

function slugName(chapter) {
  return `${String(chapter.order).padStart(2, "0")}-${chapter.id}.svg`;
}

function lerp(a, b, t) {
  return a + (b - a) * t;
}

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function isNumeric(value) {
  return typeof value === "number" && Number.isFinite(value);
}

function riskScore(variable) {
  if (variable.data_status === "verified_qualitative") {
    return qualitativeScore(variable.qualitative_position);
  }

  if (variable.data_status !== "verified_numeric") return null;

  const { baseline, boundary, high_risk: highRisk, current, direction } = variable;
  if (!isNumeric(baseline) || !isNumeric(boundary) || !isNumeric(current)) return null;

  if (direction === "lower_is_worse") {
    if (current >= boundary) {
      const span = baseline - boundary;
      if (span === 0) return 0;
      return clamp((baseline - current) / span, 0, 1);
    }

    if (isNumeric(highRisk) && current >= highRisk) {
      const span = boundary - highRisk;
      if (span === 0) return 1;
      return 1 + clamp((boundary - current) / span, 0, 1);
    }

    if (isNumeric(highRisk)) {
      const span = boundary - highRisk;
      if (span === 0) return 2;
      return 2 + Math.max(0, (highRisk - current) / span);
    }

    return 2;
  }

  if (current <= boundary) {
    const span = boundary - baseline;
    if (span === 0) return 1;
    return clamp((current - baseline) / span, 0, 1);
  }

  if (isNumeric(highRisk) && current <= highRisk) {
    const span = highRisk - boundary;
    if (span === 0) return 1;
    return 1 + clamp((current - boundary) / span, 0, 1);
  }

  if (isNumeric(highRisk)) {
    const span = highRisk - boundary;
    if (span === 0) return 2;
    return 2 + Math.max(0, (current - highRisk) / span);
  }

  return 2;
}

function qualitativeScore(position) {
  switch (position) {
    case "safe":
      return 0.5;
    case "boundary":
      return 1;
    case "risk":
      return 1.5;
    case "high_risk":
      return 2;
    case "beyond_high_risk":
      return 3;
    default:
      return 1.5;
  }
}

function xFromScore(score, maxScore) {
  const boundedScore = Math.max(0, score);
  if (boundedScore <= 1) {
    return lerp(LAYOUT.lineX1, LAYOUT.boundaryX, boundedScore);
  }

  if (boundedScore <= 2) {
    return lerp(LAYOUT.boundaryX, LAYOUT.highRiskX, boundedScore - 1);
  }

  const span = Math.max(1, maxScore - 2);
  return lerp(
    LAYOUT.highRiskX,
    LAYOUT.lineX2,
    clamp((boundedScore - 2) / span, 0, 1)
  );
}

function visibleMarkers(chapter) {
  const entries = [];
  for (const variable of chapter.variables) {
    if (variable.data_status === "placeholder") {
      console.warn(
        `[warning] ${chapter.id}/${variable.id}: placeholder marker skipped.`
      );
      continue;
    }

    if (
      variable.data_status !== "verified_numeric" &&
      variable.data_status !== "verified_qualitative"
    ) {
      continue;
    }

    const score = riskScore(variable);
    if (!isNumeric(score)) {
      console.warn(
        `[warning] ${chapter.id}/${variable.id}: marker skipped because no numeric or qualitative position was available.`
      );
      continue;
    }

    entries.push({ variable, score });
  }

  const maxScore = Math.max(3, ...entries.map((entry) => entry.score));
  return entries.map((entry) => ({
    ...entry,
    x: xFromScore(entry.score, maxScore)
  }));
}

function estimateTextWidthMm(text) {
  let width = 0;
  for (const char of text) {
    width += char.charCodeAt(0) > 127 ? 2.05 : 1.35;
  }
  return width;
}

function placeMarkerLabels(markers) {
  const placed = [];
  const sorted = markers.slice().sort((a, b) => a.x - b.x);
  for (const marker of sorted) {
    const width = estimateTextWidthMm(marker.variable.label);
    let level = 0;
    while (
      placed.some((other) => {
        if (other.level !== level) return false;
        const required = (width + other.width) / 2 + 4;
        return Math.abs(marker.x - other.x) < required;
      })
    ) {
      level += 1;
    }

    placed.push({ ...marker, width, level });
  }

  return placed;
}

function wrapText(text, maxChars) {
  if (text.length <= maxChars) return [text];
  const parts = text.split(" · ");
  const lines = [];
  let current = "";
  for (const part of parts) {
    const next = current ? `${current} · ${part}` : part;
    if (next.length > maxChars && current) {
      lines.push(current);
      current = part;
    } else {
      current = next;
    }
  }
  if (current) lines.push(current);
  return lines;
}

function fontSize(value) {
  return Number(value).toFixed(2);
}

function textLine({ x, y, text, size = FONT.note, weight = 400, anchor = "start", extra = "" }) {
  return `<text x="${x.toFixed(2)}" y="${y.toFixed(2)}" font-size="${fontSize(size)}" font-weight="${weight}" letter-spacing="0" text-anchor="${anchor}" ${extra}>${esc(text)}</text>`;
}

function renderTitle(chapter) {
  return textLine({
    x: LAYOUT.lineX1,
    y: LAYOUT.titleY,
    text: `${chapter.title_ko} / ${chapter.title_en}`,
    size: FONT.title,
    weight: 600
  });
}

function renderStaticLine(chapter) {
  const unresolved = chapter.variables.some(
    (variable) => variable.data_status === "unresolved_boundary"
  );

  const dash = unresolved ? ` stroke-dasharray="1.1 1.1"` : "";
  const lineParts = [
    `<line x1="${LAYOUT.lineX1}" y1="${LAYOUT.lineY}" x2="${LAYOUT.boundaryX}" y2="${LAYOUT.lineY}" stroke="${STYLE.safeLine}" stroke-width="${STROKE_MAIN}" stroke-linecap="butt"${dash}/>`,
    `<line x1="${LAYOUT.boundaryX}" y1="${LAYOUT.lineY}" x2="${LAYOUT.lineX2}" y2="${LAYOUT.lineY}" stroke="${STYLE.exceedLine}" stroke-width="${STROKE_MAIN}" stroke-linecap="butt"${dash}/>`
  ];

  if (!unresolved) {
    lineParts.push(
      `<line x1="${LAYOUT.boundaryX}" y1="${LAYOUT.lineY - LAYOUT.tickHalf}" x2="${LAYOUT.boundaryX}" y2="${LAYOUT.lineY + LAYOUT.tickHalf}" stroke="${STYLE.text}" stroke-width="${STROKE_AUX}"/>`,
      `<line x1="${LAYOUT.highRiskX}" y1="${LAYOUT.lineY - LAYOUT.tickHalf}" x2="${LAYOUT.highRiskX}" y2="${LAYOUT.lineY + LAYOUT.tickHalf}" stroke="${STYLE.text}" stroke-width="${STROKE_AUX}"/>`
    );
  }

  lineParts.push(
    textLine({
      x: (LAYOUT.lineX1 + LAYOUT.boundaryX) / 2,
      y: LAYOUT.zoneLabelY,
      text: "안전영역",
      size: FONT.zone,
      anchor: "middle"
    }),
    textLine({
      x: (LAYOUT.boundaryX + LAYOUT.highRiskX) / 2,
      y: LAYOUT.zoneLabelY,
      text: "행성한계",
      size: FONT.zone,
      anchor: "middle"
    }),
    textLine({
      x: (LAYOUT.highRiskX + LAYOUT.lineX2) / 2,
      y: LAYOUT.zoneLabelY,
      text: "고위험",
      size: FONT.zone,
      anchor: "middle"
    })
  );

  if (unresolved) {
    lineParts.push(
      textLine({
        x: (LAYOUT.lineX1 + LAYOUT.lineX2) / 2,
        y: LAYOUT.lineY + 8.8,
        text: "정량 경계 미확정",
        size: FONT.unresolved,
        anchor: "middle"
      })
    );
  }

  return lineParts.join("\n");
}

function renderMarkers(chapter) {
  const markers = placeMarkerLabels(visibleMarkers(chapter));
  const maxLabelLevel = markers.reduce((max, marker) => Math.max(max, marker.level), 0);
  const parts = [];

  for (const marker of markers) {
    const x = marker.x;
    const labelY = LAYOUT.markerLabelY + marker.level * LAYOUT.markerLabelGap;
    const labelX = clamp(x, LAYOUT.lineX1 + 6, LAYOUT.lineX2 - 2);
    let anchor = "middle";
    if (x > LAYOUT.lineX2 - 10) anchor = "end";
    if (x < LAYOUT.lineX1 + 10) anchor = "start";

    parts.push(
      `<line x1="${x.toFixed(2)}" y1="${(LAYOUT.lineY + 1.2).toFixed(2)}" x2="${x.toFixed(2)}" y2="${(labelY - 2.5).toFixed(2)}" stroke="${STYLE.text}" stroke-width="${STROKE_AUX}" stroke-dasharray="0.7 0.7"/>`,
      `<circle cx="${x.toFixed(2)}" cy="${LAYOUT.lineY}" r="0.72" fill="${STYLE.text}"/>`,
      textLine({
        x: labelX,
        y: labelY,
        text: marker.variable.label,
        size: FONT.marker,
        anchor,
        extra: `data-score="${marker.score.toFixed(4)}" data-x="${x.toFixed(2)}"`
      })
    );
  }

  return {
    svg: parts.join("\n"),
    maxLabelLevel,
    markers
  };
}

function renderNotes(chapter, startY) {
  const lines = [
    ...wrapText(`제어변수 / ${chapter.control_note}`, 48),
    ...wrapText(`PB / ${chapter.pb_note}`, 48)
  ];

  return lines
    .map((line, index) =>
      textLine({
        x: LAYOUT.noteX,
        y: startY + index * LAYOUT.noteLineGap,
        text: line,
        size: FONT.note
      })
    )
    .join("\n");
}

function renderFooter(meta) {
  const lines = [
    meta.axis_note,
    meta.footer_note || `기준연도 ${meta.basis_year || "2025"}`
  ].filter(Boolean);

  return lines
    .map((line, index) =>
      textLine({
        x: LAYOUT.footerX,
        y: LAYOUT.footerY + index * LAYOUT.footerLineGap,
        text: line,
        size: FONT.footer
      })
    )
    .join("\n");
}

function renderSvg(chapter, meta) {
  const markerRender = renderMarkers(chapter);
  const noteY = Math.max(
    LAYOUT.noteY,
    LAYOUT.markerLabelY + markerRender.maxLabelLevel * LAYOUT.markerLabelGap + 14.5
  );
  const desc = `${chapter.title_ko} / ${chapter.title_en}. Marker positions are calculated from baseline, boundary, current, and high_risk values; qualitative markers use qualitative_position.`;

  return `<?xml version="1.0" encoding="UTF-8"?>
<svg xmlns="http://www.w3.org/2000/svg" width="${PAGE.width}mm" height="${PAGE.height}mm" viewBox="0 0 ${PAGE.width} ${PAGE.height}" role="img" aria-labelledby="title desc">
  <title id="title">${esc(chapter.title_ko)} / ${esc(chapter.title_en)}</title>
  <desc id="desc">${esc(desc)}</desc>
  <rect width="${PAGE.width}" height="${PAGE.height}" fill="${STYLE.background}"/>
  <g font-family="'Noto Serif KR', 'Apple SD Gothic Neo', 'Noto Sans KR', serif" fill="${STYLE.text}">
${indent(renderTitle(chapter), 4)}
${indent(renderStaticLine(chapter), 4)}
${indent(markerRender.svg, 4)}
${indent(renderNotes(chapter, noteY), 4)}
${indent(renderFooter(meta), 4)}
  </g>
</svg>
`;
}

function indent(text, spaces) {
  const pad = " ".repeat(spaces);
  return text
    .split("\n")
    .filter((line) => line.trim().length > 0)
    .map((line) => `${pad}${line}`)
    .join("\n");
}

function renderIndex(data, files) {
  const cards = data.chapters
    .slice()
    .sort((a, b) => a.order - b.order)
    .map((chapter) => {
      const file = files.get(chapter.id);
      return `<a class="card" href="svg/${esc(file)}">
        <img src="svg/${esc(file)}" alt="${esc(chapter.title_ko)} / ${esc(chapter.title_en)}">
        <div class="caption">${chapter.order}. ${esc(chapter.title_ko)} / ${esc(chapter.title_en)}</div>
      </a>`;
    })
    .join("\n");

  return `<!doctype html>
<html lang="ko">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <title>행성한계 챕터 오프너 상태 도표 Preview</title>
  <style>
    :root { color: #000; background: #f4f2ef; }
    body {
      margin: 0;
      padding: 28px;
      font-family: -apple-system, BlinkMacSystemFont, "Apple SD Gothic Neo", "Noto Sans KR", sans-serif;
    }
    h1 {
      margin: 0 0 22px;
      font-size: 18px;
      font-weight: 600;
      letter-spacing: 0;
    }
    .grid {
      display: grid;
      grid-template-columns: repeat(auto-fit, minmax(420px, 1fr));
      gap: 22px;
      align-items: start;
    }
    .card {
      display: block;
      color: inherit;
      text-decoration: none;
      background: #fff;
      border: 1px solid #DDD8D4;
    }
    img {
      display: block;
      width: 100%;
      aspect-ratio: 210 / 280;
      object-fit: contain;
      background: #fff;
    }
    .caption {
      border-top: 1px solid #DDD8D4;
      padding: 10px 12px 12px;
      font-size: 13px;
      line-height: 1.45;
    }
  </style>
</head>
<body>
  <h1>행성한계 챕터 오프너 상태 도표 Preview</h1>
  <main class="grid">
${cards}
  </main>
</body>
</html>
`;
}

function main() {
  const data = readData();
  ensureDir(SVG_DIR);

  const files = new Map();
  const chapters = data.chapters.slice().sort((a, b) => a.order - b.order);
  for (const chapter of chapters) {
    const file = slugName(chapter);
    const svg = renderSvg(chapter, data.meta);
    fs.writeFileSync(path.join(SVG_DIR, file), svg, "utf8");
    files.set(chapter.id, file);

    const markers = visibleMarkers(chapter)
      .map((marker) => `${marker.variable.id}@${marker.x.toFixed(2)}mm`)
      .join(", ");
    const unresolved = chapter.variables.some(
      (variable) => variable.data_status === "unresolved_boundary"
    );
    console.log(
      `[ok] ${file}: ${markers || "no markers"}${unresolved ? " / 정량 경계 미확정" : ""}`
    );
  }

  fs.writeFileSync(INDEX_PATH, renderIndex(data, files), "utf8");
  console.log(`[ok] index.html generated`);
}

main();
